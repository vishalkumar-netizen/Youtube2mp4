Publish Frontend to GitHub Pages

  1. Create a new public repository on github.com/new (e.g., video-downloader).
  2. Save this entire HTML file as index.html and upload/push it to the main branch.
  3. Go to your repository Settings → Pages.
  4. Under Build and deployment → Branch, select main and click Save.
  5. Your live site will be available at https://your-username.github.io/video-downloader/.

Run Your Own Backend API (Optional / Unlimited)

Public APIs can occasionally hit rate limits. For guaranteed 1080p/1440p downloads with sound, run this lightweight Python server locally or deploy it free on Render / Railway / Koyeb, then select "Custom Python Backend" on the Downloader tab. 

# Save as server.py | Run: pip install fastapi uvicorn yt-dlp && uvicorn server:app --reload
import os, uuid
from fastapi import FastAPI, HTTPException
from fastapi.middleware.cors import CORSMiddleware
from fastapi.responses import FileResponse
import yt_dlp

app = FastAPI()
app.add_middleware(CORSMiddleware, allow_origins=["*"], allow_methods=["*"], allow_headers=["*"])
os.makedirs("downloads", exist_ok=True)

@app.get("/download")
def download_video(url: str, quality: str = "1080"):
    file_id = str(uuid.uuid4())[:8]
    outtmpl = f"downloads/{file_id}_%(title)s.%(ext)s"
    
    if quality == "audio":
        fmt = "bestaudio/best"
    else:
        fmt = f"bestvideo[height<={quality}]+bestaudio/best[height<={quality}]/best"

    ydl_opts = {
        "format": fmt,
        "outtmpl": outtmpl,
        "merge_output_format": "mp4" if quality != "audio" else "m4a",
        "noplaylist": True,
    }
    try:
        with yt_dlp.YoutubeDL(ydl_opts) as ydl:
            info = ydl.extract_info(url, download=True)
            filename = ydl.prepare_filename(info)
            if quality != "audio":
                filename = os.path.splitext(filename)[0] + ".mp4"
            return FileResponse(filename, filename=os.path.basename(filename))
    except Exception as e:
        raise HTTPException(status_code=400, detail=str(e))
